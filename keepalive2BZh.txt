```bash
#!/bin/bash

# ==============================
# VPN Keepalive + SA Recovery
# ==============================

# Глобальный IP VPN-сервера
VPN_PEER="GLOBAL_SERVER_IP"

# IP хоста за сервером, который проверяем
TARGET="10.200.100.1"

INTERVAL=30
PING_TIMEOUT=3

# Сколько неудачных проверок подряд считать проблемой
FAIL_THRESHOLD=3

# Не удалять SA чаще, чем раз в 10 минут
SA_DELETE_COOLDOWN=600

LOCK="/tmp/vpn_keepalive.lock"
LOG="/var/log/vpn_keepalive.log"
LOG_MAX_KB=10240

# ==============================
# Не допускаем второй экземпляр
# ==============================

if [ -f "$LOCK" ]; then
    OLD=$(cat "$LOCK" 2>/dev/null)

    if [ -n "$OLD" ] && kill -0 "$OLD" 2>/dev/null; then
        exit 0
    fi
fi

echo $$ > "$LOCK"

cleanup() {
    rm -f "$LOCK"
    exit
}

trap cleanup INT TERM EXIT

# ==============================
# Получаем локальный 10.x.x.x IP
# ==============================

get_src() {
    ip -4 -o addr show 2>/dev/null |
        awk '{print $4}' |
        cut -d/ -f1 |
        grep -E '^10\.' |
        head -1
}

# ==============================
# Ротация лога
# ==============================

rotate_log() {
    [ -f "$LOG" ] || return

    local kb
    kb=$(du -k "$LOG" 2>/dev/null | cut -f1)

    if [ "${kb:-0}" -gt "$LOG_MAX_KB" ]; then
        mv "$LOG" "${LOG}.old"
    fi
}

# ==============================
# Переменные состояния
# ==============================

FAIL_COUNT=0
LAST_SA_DELETE=0

# ==============================
# Основной цикл
# ==============================

while true; do

    SRC=$(get_src)
    TS=$(date '+%F %T')

    if [ -z "$SRC" ]; then

        echo "$TS ERR  src=unknown (no 10.x.x.x interface)" >> "$LOG"
        FAIL_COUNT=0

    elif ping -c1 -W"$PING_TIMEOUT" -I "$SRC" "$TARGET" >/dev/null 2>&1; then

        echo "$TS OK   $TARGET <- $SRC" >> "$LOG"
        FAIL_COUNT=0

    else

        FAIL_COUNT=$((FAIL_COUNT + 1))

        echo "$TS FAIL $TARGET <- $SRC (failure $FAIL_COUNT/$FAIL_THRESHOLD)" >> "$LOG"

        # Если несколько проверок подряд провалились —
        # пересоздаём IKE/IPsec SA конкретного VPN peer.
        if [ "$FAIL_COUNT" -ge "$FAIL_THRESHOLD" ]; then

            NOW=$(date +%s)

            if [ $((NOW - LAST_SA_DELETE)) -ge "$SA_DELETE_COOLDOWN" ]; then

                echo "$TS RECOVERY deleting IKE/IPsec SA for peer $VPN_PEER" >> "$LOG"

                vpn tu del "$VPN_PEER" >> "$LOG" 2>&1

                LAST_SA_DELETE=$NOW
                FAIL_COUNT=0

                echo "$TS RECOVERY SA delete completed" >> "$LOG"

            else

                echo "$TS RECOVERY cooldown active, SA not deleted" >> "$LOG"

            fi
        fi
    fi

    rotate_log
    sleep "$INTERVAL"

done
```
